function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var screenshotUrl = "No file"; if (data.screenshotData && data.screenshotData.indexOf("base64,") > -1) { try { var folderName = "VBD_Registration_Screenshots_2026"; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); var splitData = data.screenshotData.split("base64,"); var contentType = splitData[0].split(":")[1].split(";")[0]; var bytes = Utilities.base64Decode(splitData[1]); var blob = Utilities.newBlob(bytes, contentType, (data.fullName || "Delegate") + "_" + (data.screenshotName || "payment.png")); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); screenshotUrl = file.getUrl(); } catch (err) { screenshotUrl = "Upload Error: " + err.message; } } sheet.appendRow([ new Date(), data.fullName, data.phone, data.email, data.category, data.subCategory || "", data.upazila || "", data.role || "", data.institution || "", data.feeTier || "", screenshotUrl ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); }
